4. ui/script/hotkeys.lua (decompiled camera functions)
------------------------------------------------------
-- wheel zoom: multiplicative, gated by IsCameraZoomEnabled
function CameraZoomIn()
  if IsCameraZoomEnabled then Camera_Zoom(0.9) end
end
function CameraZoomOut()
  if IsCameraZoomEnabled then Camera_Zoom(1.1) end
end

-- LMB / RMB drag entry points (mode 0 = camera, mode 1 = camera+character)
function CameraOrSelectOrMoveStart(A0_2)
  Ctrl_CameraOrSelectOrMoveStart(A0_2)
end
function CameraOrSelectOrMoveStop(A0_2)
  Ctrl_CameraOrSelectOrMoveStop(A0_2)
end

-- F11 reset: camera behind the character, pitch -15 deg
function CameraReset()
  local L0_2 = GetClientPlayer()
  if not L0_2 then return end
  local L1_2 = L0_2.nFaceDirection - 64
  if L1_2 < 0 then L1_2 = L1_2 + 256 end
  local L2_2 = 256 - L1_2
  L2_2 = L2_2 / 256
  local L3_2 = math.pi
  L2_2 = L2_2 * L3_2
  L1_2 = L2_2 * 2
  Camera_SetForceReset(L1_2, math.pi / -12, 1)
end

-- CameraSetView(deg) stores a view preset (0 / 180 from bindings)
function L44_1(A0_2)
  L43_1 = A0_2
end
CameraSetView = L43_1

6. Reproduce
------------
- Extract UI files: tools/netcode/extract_pak_paths.py with a candidates list
  (ui/module_info.xml, ui/hotkey/default.txt, ui/hotkey/bindings.ini,
  ui/script/hotkeys.lua, ui/script/control.lua,
  ui/script/operationmodebase.lua, ui/script/cameracommon.lua).
- Bytecode dump: tools/netcode/lua51_dump.py FILE --code
- Decompile: java -jar unluac.jar FILE > out.lua   (unluac_2023_12_24.jar)
- Represent disasm: tools/dump_va.py JX3RepresentX64.dll 0x180b36e80 0x180b37060
